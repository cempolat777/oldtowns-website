function hashText(value: string) {
  let hash = 2166136261;

  for (let index = 0; index < value.length; index += 1) {
    hash ^= value.charCodeAt(index);
    hash = Math.imul(hash, 16777619);
  }

  return hash >>> 0;
}

function formatVideoDate(value: string | undefined, lang: string) {
  if (!value) return '';

  const date = new Date(value);
  if (!Number.isFinite(date.getTime())) return '';

  try {
    return new Intl.DateTimeFormat(lang, {
      year: 'numeric',
      month: 'long',
      day: 'numeric'
    }).format(date);
  } catch {
    return value.slice(0, 10);
  }
}

function cleanSourceTitle(value: string) {
  return String(value || '')
    .replace(/https?:\/\/\S+/gi, ' ')
    .replace(/#[\p{L}\p{N}_-]+/gu, ' ')
    .replace(/[\u{1F1E6}-\u{1F1FF}]{2}/gu, ' ')
    .replace(/[\p{Extended_Pictographic}\uFE0F]/gu, ' ')
    .replace(/[【】\[\]<>]/g, ' ')
    .replace(/\b(?:4K|8K|HDR|UHD|60FPS|60\s*FPS|ULTRA\s*HD|ASMR|NO\s*TALKING|CAPTIONS?)\b/gi, ' ')
    .replace(/[|•·]+/g, ' | ')
    .replace(/\s+/g, ' ')
    .trim()
    .slice(0, 180);
}

function normalizeComparison(value: string) {
  return String(value || '')
    .normalize('NFKC')
    .toLocaleLowerCase()
    .replace(/[.,|–—-]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

function escapeRegExp(value: string) {
  return value.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

function looksLikeDateDetail(value: string) {
  const normalized = normalizeComparison(value);

  if (!normalized) return false;

  const hasYear = /\b(?:19|20)\d{2}\b/.test(normalized);
  const numericDate = /^\d{1,2}\s+\d{1,2}\s+(?:19|20)\d{2}$/.test(normalized);
  const monthDate = /^\d{1,2}\s+[\p{L}\p{M}.]+\s+(?:19|20)\d{2}$/u.test(normalized);
  const yearMonth = /^(?:19|20)\d{2}\s+[\p{L}\p{M}.]+$/u.test(normalized);
  const monthYear = /^[\p{L}\p{M}.]+\s+(?:19|20)\d{2}$/u.test(normalized);

  return numericDate || monthDate || yearMonth || monthYear || (hasYear && normalized.split(' ').length <= 5);
}

function extractSeoDetail(
  title: string,
  publishedAt: string | undefined,
  lang: string
) {
  const parts = title
    .split('|')
    .map((item) => item.trim())
    .filter(Boolean);

  if (parts.length < 3) return '';

  const detail = parts.slice(1, -1).join(' | ').trim();
  const formattedDate = formatVideoDate(publishedAt, lang);

  if (!detail || looksLikeDateDetail(detail)) return '';

  if (
    formattedDate &&
    normalizeComparison(detail) === normalizeComparison(formattedDate)
  ) {
    return '';
  }

  return detail;
}

const detailStopWords = new Set([
  'best','ultimate','amazing','incredible','beautiful','famous','popular','top','hidden',
  'must','see','why','how','this','that','these','those','world','worlds','discover','discovery',
  'explore','exploring','experience','experiences','relaxing','cinematic','virtual','video','videos',
  'walking','walk','tour','tours','travel','city','streets','street','full','complete','true',
  'hour','hours','minute','minutes','day','days','spring','summer','autumn','winter','record','breaking',
  'snow','snowy','rain','rainy','night','evening','wet','storm','scorching','hot','long',
  'sound','immersive','caption','captions','with','and','the','in','at','of','for','a','an','to','from','via'
]);

function removeKnownLocationWords(value: string, video: Video) {
  let result = String(value || '');

  for (const literal of [video.city, video.country]) {
    const item = String(literal || '').trim();
    if (!item) continue;

    result = result.replace(
      new RegExp(`(^|[^\\p{L}\\p{N}])${escapeRegExp(item)}(?=$|[^\\p{L}\\p{N}])`, 'giu'),
      ' '
    );
  }

  return result;
}

function cleanDetailCandidate(value: string, video: Video) {
  return removeKnownLocationWords(value, video)
    .replace(/\b(?:19|20)\d{2}\b/g, ' ')
    .replace(/\b\d+(?:\.\d+)?\s*(?:h|hr|hrs|hour|hours|min|mins|minute|minutes)\b/gi, ' ')
    .replace(/\b(?:UK|USA|UAE|POV|HDR|UHD|ASMR|FPS)\b/gi, ' ')
    .replace(/\b(?:walking|walk|tour|tours|virtual|video|full|complete|captions?|immersive|sound|ultra|city)\b/gi, ' ')
    .replace(/[(){}\[\]]/g, ' ')
    .replace(/^[\s,:;|\-–—]+|[\s,:;|\-–—]+$/g, '')
    .replace(/\s+/g, ' ')
    .trim();
}

function isSafeDetailCandidate(value: string, video: Video) {
  const cleaned = String(value || '').trim();
  if (cleaned.length < 3 || cleaned.length > 64) return false;
  if (looksLikeDateDetail(cleaned)) return false;
  if (/[!?]/.test(cleaned)) return false;
  if (/^\d+$/.test(cleaned)) return false;

  const normalized = normalizeComparison(cleaned);
  if (!normalized) return false;

  const city = normalizeComparison(video.city || '');
  const country = normalizeComparison(video.country || '');
  if (normalized === city || normalized === country) return false;

  const words = normalized.split(' ').filter(Boolean);
  if (!words.length || words.length > 7) return false;

  const meaningful = words.filter(
    (word) => /[\p{L}\p{N}]/u.test(word) && !detailStopWords.has(word)
  );
  if (!meaningful.length) return false;

  const weakCount = words.filter((word) => detailStopWords.has(word)).length;
  if (weakCount >= Math.max(2, Math.ceil(words.length * 0.6))) return false;

  return /[\p{L}]/u.test(cleaned);
}

function extractAirportDetailFromTitle(title: string) {
  const source = String(title || '')
    .replace(/[\u{1F1E6}-\u{1F1FF}]{2}/gu, ' ')
    .replace(/[\p{Extended_Pictographic}\uFE0F]/gu, ' ')
    .replace(/[【】\[\]<>]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();

  const airportMatch = source.match(
    /([A-Z][\p{L}\p{M}.'’\-]+(?:\s+[A-Z][\p{L}\p{M}.'’\-]+){0,5}\s+(?:International\s+)?Airport)\b/u
  );

  if (!airportMatch?.[1]) return '';

  const airport = airportMatch[1].trim();
  const codeMatch = source.match(/\(([A-Z]{3})\)/);
  const terminalMatch = source.match(/\bTerminal\s+([0-9]+[A-Z]?)\b/i);

  let result = airport;

  if (codeMatch?.[1] && !result.includes(`(${codeMatch[1]})`)) {
    result += ` (${codeMatch[1]})`;
  }

  if (terminalMatch?.[1] && !/\bTerminal\b/i.test(result)) {
    result += ` – Terminal ${terminalMatch[1]}`;
  }

  return result;
}

function extractFromToDetail(title: string, video: Video) {
  const source = cleanSourceTitle(title);
  let match = source.match(
    /\bfrom\s+(.+?)\s+to\s+(.+?)(?=\s+(?:via|through)\b|\s+\||\s+[–—-]\s+|$)/i
  );

  if (!match) {
    const segments = source
      .split(/\s+\|\s+|\s+[–—]\s+|\s+-\s+/)
      .map((item) => item.trim())
      .filter(Boolean);

    for (const segment of segments) {
      const direct = segment.match(/^(.+?)\s+to\s+(.+?)$/i);
      if (direct) {
        match = direct;
        break;
      }
    }
  }

  if (!match) return '';

  const start = cleanDetailCandidate(match[1], video);
  const end = cleanDetailCandidate(match[2], video);

  if (!isSafeDetailCandidate(start, video) || !isSafeDetailCandidate(end, video)) {
    return '';
  }

  return `${start} → ${end}`;
}

function extractTitleDetail(title: string, video: Video) {
  const source = cleanSourceTitle(title);
  const fromTo = extractFromToDetail(title, video);
  if (fromTo) return fromTo;

  const candidates = source
    .split(/\s+\|\s+|\s+[–—]\s+|\s+-\s+/)
    .map((part) => cleanDetailCandidate(part, video))
    .filter((part) => isSafeDetailCandidate(part, video));

  candidates.sort((a, b) => {
    const aWords = a.split(/\s+/).length;
    const bWords = b.split(/\s+/).length;
    const aCaps = /^[A-Z0-9 &'’.-]+$/.test(a) ? 2 : 0;
    const bCaps = /^[A-Z0-9 &'’.-]+$/.test(b) ? 2 : 0;
    return (bCaps + Math.min(bWords, 5)) - (aCaps + Math.min(aWords, 5));
  });

  return candidates[0] || '';
}

function extractVideoSpecificDetail(
  video: Video,
  seoTitleValue: string,
  lang: string
) {
  if (video.category === 'Airport Walks') {
    const airportDetail = extractAirportDetailFromTitle(video.title);
    if (airportDetail) return airportDetail;
  }

  const seoDetail = extractSeoDetail(
    seoTitleValue,
    video.publishedAt,
    lang
  );

  if (seoDetail && isSafeDetailCandidate(seoDetail, video)) {
    return seoDetail;
  }

  const titleDetail = extractTitleDetail(video.title, video);
  if (titleDetail) return titleDetail;

  return '';
}

function extractRoutePlaces(description?: string) {
  if (!description) return [] as string[];

  const source = String(description).replace(/\r/g, ' ');
  const pattern = /(?:^|\s)(?:\d{1,2}:)?\d{1,2}:\d{2}\s*[-–—:]?\s*(.*?)(?=\s+(?:\d{1,2}:)?\d{1,2}:\d{2}\b|$)/g;
  const places: string[] = [];
  const seen = new Set<string>();
  let match: RegExpExecArray | null;

  while ((match = pattern.exec(source)) !== null) {
    const place = String(match[1] || '')
      .replace(/https?:\/\/\S+/gi, ' ')
      .replace(/[#@][\p{L}\p{N}_-]+/gu, ' ')
      .replace(/\b(?:intro|preview|outro|subscribe|chapter|chapters|start|ending)\b/gi, ' ')
      .replace(/[|•·]+/g, ' ')
      .replace(/\s+/g, ' ')
      .trim()
      .slice(0, 72);

    const key = place.toLocaleLowerCase();

    if (
      place.length >= 3 &&
      place.length <= 72 &&
      !looksLikeDateDetail(place) &&
      !seen.has(key)
    ) {
      seen.add(key);
      places.push(place);
    }

    if (places.length >= 4) break;
  }

  return places;
}

function formatRouteList(items: string[], lang: string) {
  if (!items.length) return '';

  try {
    return new Intl.ListFormat(lang, {
      style: 'long',
      type: 'conjunction'
    }).format(items);
  } catch {
    return items.join(', ');
  }
}

function normalizeNarrativeQuality(value?: string) {
  const badge = String(value || '').trim();
  if (!badge) return '';

  if (/8K/i.test(badge)) {
    return /60\s*FPS/i.test(badge) ? '8K 60FPS' : '8K';
  }

  if (/4K/i.test(badge)) {
    return /60\s*FPS/i.test(badge) ? '4K 60FPS' : '4K';
  }

  return '';
}

function getNarrativeSceneKey(video: Video) {
  const title = String(video.title || '').toLocaleLowerCase();

  if (video.category === 'Night & Rain') {
    const hasRain = /\b(?:rain|rainy|storm|wet)\b/i.test(title);
    const hasNight = /\b(?:night|evening|nighttime|after dark)\b/i.test(title);

    if (hasRain && hasNight) return 'rainNight';
    if (hasRain) return 'rain';
    if (hasNight) return 'night';
    return 'night';
  }

  switch (video.category) {
    case 'Airport Walks': return 'airport';
    case 'Beach Walking Tours': return 'beach';
    case 'Drone & Aerial': return 'drone';
    case 'Street Food': return 'food';
    case 'Nature Trails': return 'nature';
    case 'Museums & Culture': return 'culture';
    case 'POV Rides': return 'pov';
    case 'Documentaries': return 'documentary';
    default: return 'walk';
  }
}

function fillNarrative(
  template: string,
  values: Record<string, string>
) {
  return template.replace(
    /\{(\w+)\}/g,
    (_, key: string) => values[key] || ''
  );
}

function pickNarrative(
  items: string[],
  seed: number,
  offset: number
) {
  return items[(seed + offset) % items.length];
}

function buildSourceFact(context: NarrativeContext) {
  const pack = sourceFactText[context.lang] || sourceFactText.en;
  const template = context.date && context.quality
    ? pack.both
    : context.date
      ? pack.date
      : context.quality
        ? pack.quality
        : pack.basic;

  return fillNarrative(template, {
    source: context.source,
    date: context.date,
    quality: context.quality
  });
}

function buildNarrative(context: NarrativeContext) {
  const pack = naturalNarrativeText[context.lang] || naturalNarrativeText.en;
  const sourcePack = naturalSourceText[context.lang] || naturalSourceText.en;
  const values = {
    location: context.location,
    category: context.category,
    scene: context.scene,
    detail: context.detail,
    route: context.route,
    source: context.source,
    date: context.date,
    quality: context.quality || ''
  };

  const introTemplate = pack.intro[context.mode] || pack.intro.walk;
  const sourceTemplate = context.date ? sourcePack.date : sourcePack.basic;
  const paragraphs = [
    fillNarrative(introTemplate, values),
    context.route ? fillNarrative(pack.route, values) : '',
    fillNarrative(sourceTemplate, values)
  ];

  return paragraphs
    .map((paragraph) => paragraph.replace(/\s+/g, ' ').replace(/\s+([,.;:!?])/g, '$1').trim())
    .filter(Boolean);
}
